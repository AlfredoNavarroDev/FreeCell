# Reseñas de producto (S11) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** El cliente valora productos de pedidos entregados desde `/pedidos`, el admin modera en `/admin/soporte` (pestaña «Reseñas») y las reseñas aprobadas se muestran en la ficha del producto y como promedio en las tarjetas del catálogo.

**Architecture:** Módulo NestJS `src/reviews/` (cliente + admin) más dos endpoints de lectura pública en `CatalogController`. La regla «solo compradores reales» vive en un trigger de Postgres (`trg_review_requires_delivered`) y el promedio en la vista `v_product_ratings`. El frontend copia el patrón de reclamos y soporte: Server Actions por feature, `/pedidos` extendido y cola de moderación en `/admin/soporte`.

**Tech Stack:** NestJS 11 + TypeORM (SQL crudo vía `DataSource`), PostgreSQL 16, Jest + supertest (e2e), Next.js 16 (App Router) + Tailwind v4, Vitest.

**Spec:** `docs/superpowers/specs/2026-11-15-resenas-s11-design.md`

```ts
/** Pedido propio ya DELIVERED con su licencia vendida (mismo camino que usan los e2e de reclamos). */
export async function insertDeliveredOrder(
  ds: DataSource,
  input: {
    userId: string;
    planId: string;
    licenseId: string;
    currency?: Currency;
    unitPriceCents?: number;
  },
): Promise<{ orderId: string; orderItemId: string }> {
  const { orderId } = await insertOrder(ds, {
    userId: input.userId,
    currency: input.currency ?? 'PEN',
    lines: [
      {
        planId: input.planId,
        unitPriceCents: input.unitPriceCents ?? 1000,
        licenseId: input.licenseId,
      },
    ],
  });
  await ds.query(`UPDATE orders SET status = 'IN_REVIEW' WHERE id = $1`, [
    orderId,
  ]);
  await ds.query(`UPDATE orders SET status = 'DELIVERED' WHERE id = $1`, [
    orderId,
  ]);
  await ds.query(
    `UPDATE license_items SET status = 'SOLD', "soldAt" = now() WHERE id = $1`,
    [input.licenseId],
  );
  const [{ orderItemId }]: { orderItemId: string }[] = await ds.query(
    `SELECT "orderItemId" FROM license_items WHERE id = $1`,
    [input.licenseId],
  );
  return { orderId, orderItemId };
}

/** Inserta una reseña por SQL (pasa por el trigger de compradores reales). Devuelve su id. */
export async function insertReview(
  ds: DataSource,
  input: {
    userId: string;
    orderId: string;
    productId: string;
    rating: number;
    status?: ReviewStatus;
    body?: string;
  },
): Promise<string> {
  const [{ id }]: { id: string }[] = await ds.query(
    `INSERT INTO reviews ("productId", "userId", "orderId", rating, body, status)
     VALUES ($1, $2, $3, $4, $5, $6) RETURNING id`,
    [
      input.productId,
      input.userId,
      input.orderId,
      input.rating,
      input.body ?? 'Reseña de prueba con texto suficiente',
      input.status ?? ReviewStatus.PENDING,
    ],
  );
  return id;
}
```

```ts
import type { MigrationInterface, QueryRunner } from 'typeorm';

/**
 * D33/D57: «solo reseñas reales» se garantiza en Postgres, no solo en NestJS, y el promedio público
 * sale de una vista (mismo patrón que v_plan_stock) para que no pueda desincronizarse.
 */
export class ReviewFunctions1794751829384 implements MigrationInterface {
  name = 'ReviewFunctions1794751829384';

  public async up(queryRunner: QueryRunner): Promise<void> {
    await queryRunner.query(
      `CREATE INDEX "IDX_reviews_approved_by_product" ON "reviews" ("productId", "createdAt") WHERE "status" = 'APPROVED'`,
    );

    await queryRunner.query(`
      CREATE OR REPLACE FUNCTION fn_review_requires_delivered()
      RETURNS trigger LANGUAGE plpgsql AS $$
      BEGIN
        IF NOT EXISTS (
          SELECT 1
            FROM orders o
            JOIN order_items oi ON oi."orderId" = o.id
            JOIN plans pl ON pl.id = oi."planId"
           WHERE o.id = NEW."orderId"
             AND o."userId" = NEW."userId"
             AND o.status = 'DELIVERED'
             AND pl."productId" = NEW."productId"
        ) THEN
          RAISE EXCEPTION 'Solo puedes reseñar un producto de un pedido tuyo entregado' USING ERRCODE = 'P0001';
        END IF;
        RETURN NEW;
      END $$;
    `);
    await queryRunner.query(`
      CREATE TRIGGER trg_review_requires_delivered BEFORE INSERT ON reviews
      FOR EACH ROW EXECUTE FUNCTION fn_review_requires_delivered()
    `);

    await queryRunner.query(`
      CREATE VIEW v_product_ratings AS
      SELECT "productId",
             ROUND(AVG(rating)::numeric, 2)::numeric(3,2) AS "ratingAvg",
             COUNT(*)::int AS "ratingCount"
        FROM reviews
       WHERE status = 'APPROVED'
       GROUP BY "productId"
    `);
  }

  public async down(queryRunner: QueryRunner): Promise<void> {
    await queryRunner.query(`DROP VIEW IF EXISTS v_product_ratings`);
    await queryRunner.query(
      `DROP TRIGGER IF EXISTS trg_review_requires_delivered ON reviews`,
    );
    await queryRunner.query(
      `DROP FUNCTION IF EXISTS fn_review_requires_delivered()`,
    );
    await queryRunner.query(`DROP INDEX IF EXISTS "IDX_reviews_approved_by_product"`);
  }
}
```

```ts
import {
  ConflictException,
  ForbiddenException,
  Injectable,
  NotFoundException,
} from '@nestjs/common';
import { DataSource } from 'typeorm';
import { AuditService } from '../audit/audit.service';
import { PaginatedResponseDto } from '../common/dto/paginated-response.dto';
import type { AuthenticatedUser } from '../common/types/authenticated-user';
import { OrderStatus } from '../database/enums';
import { CreateReviewDto } from './dto/create-review.dto';
import { ListReviewsQueryDto } from './dto/list-reviews-query.dto';
import { ReviewResponseDto } from './dto/review-response.dto';

const REVIEW_COLUMNS = `r.id, r."orderId", r."productId", pr.name AS "productName",
              r.rating, r.body, r.status, r."createdAt"`;

@Injectable()
export class ReviewsService {
  constructor(
    private readonly ds: DataSource,
    private readonly audit: AuditService,
  ) {}

  /** RF-CAT-10/D33: pedido propio entregado que contenga el producto. El FOR UPDATE sobre el pedido
   * serializa reseñas simultáneas del mismo producto; el trigger de Postgres es la red de seguridad. */
  async create(
    user: AuthenticatedUser,
    dto: CreateReviewDto,
  ): Promise<ReviewResponseDto> {
    const reviewId = await this.ds.transaction(async (m) => {
      const orders: { status: OrderStatus }[] = await m.query(
        `SELECT status FROM orders WHERE id = $1 AND "userId" = $2 FOR UPDATE`,
        [dto.orderId, user.id],
      );
      const order = orders[0];
      if (!order) {
        throw new NotFoundException('Pedido no encontrado');
      }
      if (order.status !== OrderStatus.DELIVERED) {
        throw new ForbiddenException(
          'Solo puedes valorar un producto de un pedido entregado',
        );
      }

      const inOrder: unknown[] = await m.query(
        `SELECT 1
           FROM order_items oi
           JOIN plans pl ON pl.id = oi."planId"
          WHERE oi."orderId" = $1 AND pl."productId" = $2
          LIMIT 1`,
        [dto.orderId, dto.productId],
      );
      if (inOrder.length === 0) {
        throw new ForbiddenException('Ese producto no es parte de este pedido');
      }

      const existing: unknown[] = await m.query(
        `SELECT 1 FROM reviews WHERE "orderId" = $1 AND "productId" = $2`,
        [dto.orderId, dto.productId],
      );
      if (existing.length > 0) {
        throw new ConflictException('Ya valoraste este producto en este pedido');
      }

      const [{ id }]: { id: string }[] = await m.query(
        `INSERT INTO reviews ("productId", "userId", "orderId", rating, body)
         VALUES ($1, $2, $3, $4, $5) RETURNING id`,
        [dto.productId, user.id, dto.orderId, dto.rating, dto.body],
      );

      await this.audit.log(m, {
        actorId: user.id,
        action: 'review.created',
        entity: 'review',
        entityId: id,
        meta: {
          orderId: dto.orderId,
          productId: dto.productId,
          rating: dto.rating,
        },
      });
      return id;
    });

    return this.findOne(user.id, reviewId);
  }

  async listForUser(
    userId: string,
    dto: ListReviewsQueryDto,
  ): Promise<PaginatedResponseDto<ReviewResponseDto>> {
    const offset = (dto.page - 1) * dto.pageSize;
    const rows: ReviewResponseDto[] = await this.ds.query(
      `SELECT ${REVIEW_COLUMNS}
         FROM reviews r
         JOIN products pr ON pr.id = r."productId"
        WHERE r."userId" = $1
        ORDER BY r."createdAt" DESC, r.id
        LIMIT $2 OFFSET $3`,
      [userId, dto.pageSize, offset],
    );
    const [{ count }]: { count: string }[] = await this.ds.query(
      `SELECT count(*) FROM reviews WHERE "userId" = $1`,
      [userId],
    );
    return PaginatedResponseDto.from(
      rows,
      Number(count),
      dto.page,
      dto.pageSize,
    );
  }

  private async findOne(
    userId: string,
    reviewId: string,
  ): Promise<ReviewResponseDto> {
    const rows: ReviewResponseDto[] = await this.ds.query(
      `SELECT ${REVIEW_COLUMNS}
         FROM reviews r
         JOIN products pr ON pr.id = r."productId"
        WHERE r.id = $1 AND r."userId" = $2`,
      [reviewId, userId],
    );
    return rows[0];
  }
}
```

```ts
/** D33: nunca se publica el apellido completo ni el correo: «Carlos P.» o «Cliente verificado». */
export function formatAuthorName(name: string | null | undefined): string {
  const parts = (name ?? '').trim().split(/\s+/).filter(Boolean);
  if (parts.length === 0) {
    return 'Cliente verificado';
  }
  if (parts.length === 1) {
    return parts[0];
  }
  const initial = Array.from(parts[parts.length - 1])[0].toLocaleUpperCase(
    'es-PE',
  );
  return `${parts[0]} ${initial}.`;
}
```

```ts
import {
  ConflictException,
  Injectable,
  NotFoundException,
} from '@nestjs/common';
import { DataSource } from 'typeorm';
import { AuditService } from '../audit/audit.service';
import { PaginatedResponseDto } from '../common/dto/paginated-response.dto';
import type { AuthenticatedUser } from '../common/types/authenticated-user';
import { ReviewStatus } from '../database/enums';
import {
  AdminReviewResponseDto,
  ReviewActionResultDto,
} from './dto/admin-review-response.dto';
import { ListAdminReviewsQueryDto } from './dto/list-admin-reviews-query.dto';

@Injectable()
export class AdminReviewsService {
  constructor(
    private readonly ds: DataSource,
    private readonly audit: AuditService,
  ) {}

  /** Cola de moderación: la más antigua primero (igual que reclamos). */
  async list(
    dto: ListAdminReviewsQueryDto,
  ): Promise<PaginatedResponseDto<AdminReviewResponseDto>> {
    const offset = (dto.page - 1) * dto.pageSize;
    const rows: AdminReviewResponseDto[] = await this.ds.query(
      `SELECT r.id AS "reviewId", r.status, r.rating, r.body, pr.name AS "productName",
              o."orderNumber", u.name AS "customerName", u.email AS "customerEmail",
              r."createdAt"
         FROM reviews r
         JOIN products pr ON pr.id = r."productId"
         JOIN orders o ON o.id = r."orderId"
         JOIN users u ON u.id = r."userId"
        WHERE ($1::text IS NULL OR r.status::text = $1)
        ORDER BY r."createdAt" ASC, r.id
        LIMIT $2 OFFSET $3`,
      [dto.status ?? null, dto.pageSize, offset],
    );
    const [{ count }]: { count: string }[] = await this.ds.query(
      `SELECT count(*) FROM reviews WHERE ($1::text IS NULL OR status::text = $1)`,
      [dto.status ?? null],
    );
    return PaginatedResponseDto.from(
      rows,
      Number(count),
      dto.page,
      dto.pageSize,
    );
  }

  approve(admin: AuthenticatedUser, id: string): Promise<ReviewActionResultDto> {
    return this.moderate(admin, id, ReviewStatus.APPROVED, 'review.approved');
  }

  hide(admin: AuthenticatedUser, id: string): Promise<ReviewActionResultDto> {
    return this.moderate(admin, id, ReviewStatus.HIDDEN, 'review.hidden');
  }

  /** Cualquier cambio entre los tres estados es válido salvo repetir el actual (409). El FOR UPDATE
   * serializa moderaciones simultáneas de la misma reseña. */
  private async moderate(
    admin: AuthenticatedUser,
    id: string,
    target: ReviewStatus,
    action: string,
  ): Promise<ReviewActionResultDto> {
    return this.ds.transaction(async (m) => {
      const rows: { status: ReviewStatus }[] = await m.query(
        `SELECT status FROM reviews WHERE id = $1 FOR UPDATE`,
        [id],
      );
      const current = rows[0]?.status;
      if (!current) {
        throw new NotFoundException('Reseña no encontrada');
      }
      if (current === target) {
        throw new ConflictException('La reseña ya tiene ese estado');
      }

      await m.query(`UPDATE reviews SET status = $2 WHERE id = $1`, [
        id,
        target,
      ]);
      await this.audit.log(m, {
        actorId: admin.id,
        action,
        entity: 'review',
        entityId: id,
        meta: { previousStatus: current },
      });
      return { reviewId: id, status: target };
    });
  }
}
```

```ts
"use server";

import { auth } from "@/auth";

const API_URL = process.env.API_URL ?? "http://localhost:3000";

export type ReviewStatus = "PENDING" | "APPROVED" | "HIDDEN";

export interface MyReviewView {
  id: string;
  orderId: string;
  productId: string;
  productName: string;
  rating: number;
  body: string;
  status: ReviewStatus;
  createdAt: string;
}

export interface PaginatedMyReviews {
  items: MyReviewView[];
  total: number;
  page: number;
  pageSize: number;
}

export type ReviewsActionResult<T> =
  | { ok: true; data: T }
  | { ok: false; status: number; message: string };

async function callBackend<T>(
  path: string,
  init: { method: "GET" | "POST"; body?: unknown },
): Promise<ReviewsActionResult<T>> {
  const session = await auth();
  if (!session?.accessToken) {
    return { ok: false, status: 401, message: "Inicia sesión para continuar" };
  }

  const res = await fetch(`${API_URL}${path}`, {
    method: init.method,
    headers: {
      "Content-Type": "application/json",
      Authorization: `Bearer ${session.accessToken}`,
    },
    body: init.body ? JSON.stringify(init.body) : undefined,
    cache: "no-store",
  });

  if (!res.ok) {
    const body = (await res.json().catch(() => null)) as { message?: string | string[] } | null;
    const message = Array.isArray(body?.message)
      ? body.message.join(", ")
      : body?.message ?? "No pudimos completar la operación";
    return { ok: false, status: res.status, message };
  }

  const data = (await res.json()) as T;
  return { ok: true, data };
}

export async function createReview(
  orderId: string,
  productId: string,
  rating: number,
  body: string,
): Promise<ReviewsActionResult<MyReviewView>> {
  return callBackend<MyReviewView>("/reviews", {
    method: "POST",
    body: { orderId, productId, rating, body },
  });
}

export async function listMyReviews(): Promise<ReviewsActionResult<PaginatedMyReviews>> {
  return callBackend<PaginatedMyReviews>("/reviews?pageSize=50", { method: "GET" });
}
```

```ts
"use server";

import { auth } from "@/auth";
import type { ReviewStatus } from "@/lib/reviews-client";

const API_URL = process.env.API_URL ?? "http://localhost:3000";

export interface AdminReviewView {
  reviewId: string;
  status: ReviewStatus;
  rating: number;
  body: string;
  productName: string;
  orderNumber: number;
  customerName: string | null;
  customerEmail: string;
  createdAt: string;
}

export interface AdminReviewsPage {
  items: AdminReviewView[];
  total: number;
  page: number;
  pageSize: number;
}

export interface ReviewActionResult {
  reviewId: string;
  status: ReviewStatus;
}

export type AdminReviewsActionResult<T> =
  | { ok: true; data: T }
  | { ok: false; status: number; message: string };

async function callBackend<T>(
  path: string,
  init: { method: "GET" | "POST" },
): Promise<AdminReviewsActionResult<T>> {
  const session = await auth();
  if (!session?.accessToken) {
    return { ok: false, status: 401, message: "Inicia sesión para continuar" };
  }

  const res = await fetch(`${API_URL}${path}`, {
    method: init.method,
    headers: {
      "Content-Type": "application/json",
      Authorization: `Bearer ${session.accessToken}`,
    },
    cache: "no-store",
  });

  if (!res.ok) {
    const body = (await res.json().catch(() => null)) as { message?: string | string[] } | null;
    const message = Array.isArray(body?.message)
      ? body.message.join(", ")
      : body?.message ?? "No pudimos completar la operación";
    return { ok: false, status: res.status, message };
  }

  const data = (await res.json()) as T;
  return { ok: true, data };
}

export async function fetchAdminReviews(
  status: ReviewStatus = "PENDING",
  pageSize = 50,
): Promise<AdminReviewsActionResult<AdminReviewsPage>> {
  return callBackend<AdminReviewsPage>(`/admin/reviews?status=${status}&pageSize=${pageSize}`, {
    method: "GET",
  });
}

export async function approveReview(reviewId: string): Promise<AdminReviewsActionResult<ReviewActionResult>> {
  return callBackend<ReviewActionResult>(`/admin/reviews/${reviewId}/approve`, { method: "POST" });
}

export async function hideReview(reviewId: string): Promise<AdminReviewsActionResult<ReviewActionResult>> {
  return callBackend<ReviewActionResult>(`/admin/reviews/${reviewId}/hide`, { method: "POST" });
}
```

```tsx
  approveReview,

  hideReview,
  type AdminReviewView,

function formatDate(iso: string): string {
  return new Date(iso).toLocaleDateString("es-PE", { dateStyle: "medium", timeZone: "America/Lima" });
}

export function AdminReviewsQueue({ initialReviews }: { initialReviews: AdminReviewView[] }) {
  const [reviews, setReviews] = useState(initialReviews);
  const [selectedId, setSelectedId] = useState<string | null>(initialReviews[0]?.reviewId ?? null);
  const [message, setMessage] = useState<string | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [submitting, setSubmitting] = useState(false);

  const selected = reviews.find((r) => r.reviewId === selectedId) ?? null;

  async function refetchQueue(): Promise<AdminReviewView[]> {
    const result = await fetchAdminReviews("PENDING");
    if (!result.ok) {
      setError(result.message);
      return reviews;
    }
    setReviews(result.data.items);
    return result.data.items;
  }

  async function act(kind: "approve" | "hide") {
    if (!selected) return;
    setSubmitting(true);
    setError(null);
    setMessage(null);
    const result = await (kind === "approve" ? approveReview : hideReview)(selected.reviewId);
    setSubmitting(false);
    if (!result.ok) {
      setError(result.message);
      await refetchQueue();
      return;
    }
    const fresh = await refetchQueue();
    setSelectedId(fresh[0]?.reviewId ?? null);
    setMessage(
      kind === "approve"
        ? `Reseña de ${selected.productName} aprobada y publicada.`
        : `Reseña de ${selected.productName} oculta.`,
    );
  }

  return (
    <div className="flex flex-col gap-4">
      {message ? (
        <div className="rounded-[10px] bg-success-bg px-4 py-3 font-medium text-success-ink">{message}</div>
      ) : null}
      {error ? (
        <div className="rounded-[10px] bg-danger-bg px-4 py-3 font-medium text-danger-ink">{error}</div>

          <p className="font-display text-xl font-bold text-ink">Sin reseñas pendientes</p>
          <p className="mt-1 text-muted">Cuando un cliente valore una compra, aparecerá aquí para moderarla.</p>
        </div>

        <>
          <p className="text-muted">{reviews.length} pendientes</p>
          <div className="flex flex-wrap items-start gap-5">
            <section aria-label="Cola de reseñas" className="flex w-full max-w-[360px] flex-col gap-2">
              {reviews.map((r) => {
                const active = r.reviewId === selectedId;
                return (
                  <button
                    key={r.reviewId}
                    type="button"
                    aria-pressed={active}
                    onClick={() => {
                      setSelectedId(r.reviewId);
                      setError(null);
                      setMessage(null);
                    }}
                    className={`flex min-h-11 flex-col gap-1 rounded-xl border-2 px-4 py-3 text-left ${
                      active ? "border-accent bg-accent-soft-bg" : "border-line-soft bg-surface"
                    }`}
                  >
                    <span className="font-semibold text-ink">{r.productName}</span>
                    <StarRating value={r.rating} />
                    <span className="text-sm text-muted">{r.customerName ?? r.customerEmail}</span>

              >
                <div className="mb-4">

                  <p className="text-sm text-muted">
                    {selected.customerName ?? selected.customerEmail} · Pedido #{selected.orderNumber} ·{" "}

                  </p>
                </div>
                <div className="mb-4">
                  <StarRating value={selected.rating} />

                <div className="mb-5 rounded-[10px] bg-bg px-4 py-3">
                  <p className="text-sm text-muted">Lo que escribió el cliente</p>
```
